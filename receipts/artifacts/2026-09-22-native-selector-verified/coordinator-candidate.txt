// Select a completed turn, or return to latest with an empty line.
// preflight-exec: forbidden
// preludes: form-stdlib/core.fk form-stdlib/form-cli-turn-evidence-live.fk
section [form.bml] {
 class TurnEvidenceTarget {
  def fctet-byte?(byte) = or(and(ge(byte,48),le(byte,57)),
   or(and(ge(byte,65),le(byte,90)),
    or(and(ge(byte,97),le(byte,122)),
     or(eq(byte,45),or(eq(byte,95),or(eq(byte,46),eq(byte,58)))))));
  def fctet-valid?(text,at) = if ge(at,str_len(text)) then 1
   else if fctet-byte?(str_byte_at(text,at)) then fctet-valid?(text,add(at,1)) else 0;
  def fctet-bind(turnId) {
   write_file_text(fctel-target-turn-path(),turnId);
   write_file_text(fctel-discovery-path(),"");
   write_file_text(fctel-progress-path(),"");
   write_file_text(fctel-start-discovery-path(),"");
   write_file_text(fctel-candidate-row-path(),"");
   write_file_text(fctel-latest-validation-path(),"");
   print_str(if eq(str_len(turnId),0) then "turn-evidence-target=latest\n" else "turn-evidence-target=historical\n");
   0;
  }
 }
 let turnId = read_line();
 if nothing?(turnId) then form_error("turn-evidence-target: supply an empty line for latest or a raw turn identifier; EOF leaves collector state unchanged")
 else if fctet-valid?(turnId,0) then fctet-bind(turnId)
 else form_error("turn-evidence-target: expected only ASCII letters, digits, hyphen, underscore, dot or colon; collector state unchanged");
}
