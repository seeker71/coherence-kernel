  def fcap-syntax-apply(s,raw,checker,error,candidate) {
   let evidence = str_concat("JSON syntax failure: ",json-emit(faj-diagnostic(raw,error)));
   let counted = fcap-with-memory(s,list(add(fcap-failures(s),1),evidence,fcap-reflections(s),fcap-checks(s),0));
   // The repaired command enters the ordinary role, writable-path, guarded
   // edit and caller-check boundaries. Grammar repair grants no authority.
   let next = fcap-response(counted,candidate,checker);
   let inserted = gt(str_len(candidate),str_len(raw));
   let at = faj-closer-offset(raw,error);
   let method = if inserted then "native-json-missing-closer-insertion" else if eq(faj-error-reason(error),13) then "native-json-premature-root-closer-removal" else "native-json-mismatched-closer-removal";
   let row = json-node-object(list(fcap-shape-s("method",method),fcap-shape-n("byte_offset",at),fcap-shape-n(if inserted then "inserted_byte" else "removed_byte",fat-char(if inserted then candidate else raw,at)),fcap-shape-s("before_sha256",hex-encode(sha256-stream-string(raw))),fcap-shape-s("after_sha256",hex-encode(sha256-stream-string(candidate))),fcap-shape-n("strict_candidate_valid",1),fcap-shape-n("tool_calls",sub(fcap-calls(next),fcap-calls(s))),fcap-shape-n("checks",sub(fcap-checks(next),fcap-checks(s))),fcap-shape-s("result_stage",fcap-stage(next))));
   let saved = fcap-with-json-repairs(next,append(fcap-json-repairs(next),list(json-emit(row))));
   let note = json-emit(json-node-object(list(json-node-pair("native_json_repair",row),json-node-pair("observation",if faj-valid(fcap-note(next)) then parse-json("native-json-tool-result",fcap-note(next)) else json-node-string(fcap-note(next))))));
   fcap-inherit(saved,fcap-state(fcap-stage(saved),nth(saved,1),nth(saved,2),fcap-docs(saved),nth(saved,4),nth(saved,5),nth(saved,6),nth(saved,7),note,fcap-turns(saved),fcap-calls(saved),nth(saved,11)));
  }
  def fcap-syntax-revise(s,raw,end) {
   let evidence = str_concat("JSON syntax failure: ",json-emit(faj-diagnostic(raw,end)));
   let repeated = str_eq(evidence,fcap-evidence(s));
   let note = str_concat(evidence,if repeated then " The same syntax failure repeated. Correct the named byte in the original response and return one complete JSON object. No tool ran and no document changed." else " Correct this syntax in the original response and return one complete JSON object. No tool ran and no document changed.");
   let revised = fcap-revise(s,note);
   fcap-with-memory(revised,list(add(fcap-failures(s),1),evidence,fcap-reflections(s),fcap-checks(s),repeated));
  }
