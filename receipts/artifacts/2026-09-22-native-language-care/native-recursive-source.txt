// form-cli-turn-evidence-target-bind.fk — optionally select one completed turn
// for a historical Form-native re-witness. An empty line returns to latest.
//
// preflight-exec: forbidden
// preludes: form-stdlib/bml/core.bml form-stdlib/bml/form-cli-turn-evidence-live.bml

section [form.bml] {
  let turn-id = read_line();
  if nothing?(turn-id) then form_error("turn-evidence-target: expected one line, got EOF") else 0;
  if str_len(turn-id) == 0 then 0 else 0;
  def valid-chars(text, i) = if i >= str_len(text) then 1 else and(str_find("0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz-_.:", substring(text, i, i + 1), 0) >= 0, valid-chars(text, i + 1));
  if not(valid-chars(turn-id, 0)) then form_error("turn-evidence-target: invalid identifier; allowed: ASCII letters, digits, hyphen, underscore, dot, colon") else 0;
  write_file_text(fctel-target-turn-path(), turn-id);
  write_file_text(fctel-discovery-path(), "");
  write_file_text(fctel-progress-path(), "");
  write_file_text(fctel-start-discovery-path(), "");
  write_file_text(fctel-candidate-row-path(), "");
  write_file_text(fctel-latest-validation-path(), "");
  print_str(if file_size(fctel-target-turn-path()) == 0 then "turn-evidence-target=latest\n" else "turn-evidence-target=historical\n");
  0;
}
